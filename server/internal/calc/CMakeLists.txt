include(GolangTestUtils)

provision_golang_tests(${CMAKE_CURRENT_SOURCE_DIR})

# fourier_twiddles_amd64.s has to come from an amd64 build of gen/main.go that
# doesn't use FMA. That has nothing to do with the hand written assembly, it's
# because the tables come from math.Sincos, which is regular Go. On arm64, or
# on amd64 with GOAMD64=v3, Go turns its multiplies and adds into FMA
# instructions, and that changes the last bit of some of the table entries.
set(FOURIER_TABLES_ENV GOARCH=amd64 GOAMD64=v1)
set(FOURIER_TABLES_NATIVE_HOST OFF)
if(CMAKE_HOST_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
  set(FOURIER_TABLES_NATIVE_HOST ON)
endif()

if(CMAKE_Go_COMPILER)
  # Regenerates fourier_twiddles_amd64.s with those settings, this is what
  # `make fourier-tables` runs. A host that isn't x86_64 can only run the amd64
  # generator through an emulator, so it warns about that first.
  set(FOURIER_TABLES_WARNING)
  if(NOT FOURIER_TABLES_NATIVE_HOST)
    set(FOURIER_TABLES_WARNING
      COMMAND ${CMAKE_COMMAND} -E echo "WARNING: This host is ${CMAKE_HOST_SYSTEM_PROCESSOR}, but the fourier tables have to be generated by an amd64 build. It can only run through an emulator like Rosetta or qemu, so if it fails regenerate the tables on an x86_64 machine instead."
    )
  endif()
  add_custom_target(
    generate.fourier.tables
    ${FOURIER_TABLES_WARNING}
    COMMAND ${CMAKE_COMMAND} -E env ${FOURIER_TABLES_ENV} ${CMAKE_Go_COMPILER} generate ${CMAKE_CURRENT_SOURCE_DIR}/fourier.go
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    VERBATIM
  )
endif()

# If calc.fourier.tables fails that means fourier_twiddles_amd64.s no longer
# matches what gen/main.go produces, and you need to run `make fourier-tables`.
# calc.fourier.tables.generate runs first and regenerates the tables into the
# build directory with the same arguments as the go:generate line in
# fourier.go, so the checked in file is never touched. Hosts that aren't x86_64
# skip both since they can't run the amd64 generator natively.
if(BUILD_TESTING AND CMAKE_Go_COMPILER AND FOURIER_TABLES_NATIVE_HOST)
  set(FOURIER_TABLES ${CMAKE_CURRENT_BINARY_DIR}/fourier_twiddles_amd64.s)
  add_test(
    NAME calc.fourier.tables.generate
    COMMAND ${CMAKE_Go_COMPILER} run ./gen -size 4096 -output ${FOURIER_TABLES}
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  )
  set_tests_properties(calc.fourier.tables.generate PROPERTIES
    ENVIRONMENT "${FOURIER_TABLES_ENV}"
    FIXTURES_SETUP fourier.tables
    LABELS "server"
  )

  add_test(
    NAME calc.fourier.tables
    COMMAND diff -u ${CMAKE_CURRENT_SOURCE_DIR}/fourier_twiddles_amd64.s ${FOURIER_TABLES}
  )
  set_tests_properties(calc.fourier.tables PROPERTIES
    FIXTURES_REQUIRED fourier.tables
    LABELS "server"
  )
endif()
